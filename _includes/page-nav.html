{%- comment -%} Previous/next links within the current collection. just-the-docs has breadcrumbs but no sequential navigation, and the docs now have a reading order worth following (concepts especially). This walks the current collection in the same order the sidebar shows: top-level pages by nav_order, each followed by its children. Included from _layouts/default.html, which overrides the theme layout only to add this and nothing else. {%- endcomment -%} {%- assign coll = page.collection -%} {%- if coll and coll != "posts" -%} {%- assign docs = site[coll] | where_exp: "d", "d.title" -%} {%- comment -%} Top-level pages of this collection, in nav order. {%- endcomment -%} {%- assign tops = docs | where_exp: "d", "d.parent == nil" | sort: "nav_order" -%} {%- comment -%} Flatten: each top-level page, then its children in nav order. {%- endcomment -%} {%- assign flat = "" | split: "" -%} {%- for t in tops -%} {%- assign flat = flat | push: t -%} {%- assign kids = docs | where: "parent", t.title | sort: "nav_order" -%} {%- for k in kids -%} {%- assign flat = flat | push: k -%} {%- assign grandkids = docs | where: "parent", k.title | sort: "nav_order" -%} {%- for g in grandkids -%} {%- assign flat = flat | push: g -%} {%- endfor -%} {%- endfor -%} {%- endfor -%} {%- assign urls = flat | map: "url" -%} {%- assign here = -1 -%} {%- for u in urls -%} {%- if u == page.url -%}{%- assign here = forloop.index0 -%}{%- endif -%} {%- endfor -%} {%- if here >= 0 and flat.size > 1 -%} {%- assign prev_i = here | minus: 1 -%} {%- assign next_i = here | plus: 1 -%} {%- assign prev = nil -%} {%- assign next = nil -%} {%- if here > 0 -%}{%- assign prev = flat[prev_i] -%}{%- endif -%} {%- if next_i < flat.size -%}{%- assign next = flat[next_i] -%}{%- endif -%} {%- if prev or next %}
{%- if prev %}
Previous {{ prev.title }}
{%- endif %} {%- if next %}
Next {{ next.title }}
{%- endif %}
{%- endif -%} {%- endif -%} {%- endif -%}
